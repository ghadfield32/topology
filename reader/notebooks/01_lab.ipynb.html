from shape_lab.algebra import rank_mod2
A = np.array([[1, 1, 0], [1, 0, 1], [0, 1, 1]])
real_rank = int(np.linalg.matrix_rank(A))
binary_rank = rank_mod2(A)
print("Real rank:", real_rank, "F2 rank:", binary_rank)
assert real_rank == 3 and binary_rank == 2
B = np.array([[1, 1], [0, 0]])
kernel_vector = np.array([1, -1])
print("Known kernel vector mapped to:", B @ kernel_vector)
assert np.array_equal(B @ kernel_vector, [0, 0])

Known kernel vector mapped to: [0 0]

Real data: measurement columns and rank

Four columns specify an ambient representation. Matrix rank alone does not prove that the biological population is a manifold.

from shape_lab.data import load_iris_data
iris = load_iris_data()
columns = ["sepal_length_cm", "sepal_width_cm", "petal_length_cm", "petal_width_cm"]
X = iris[columns].to_numpy()
print("Species labels:", sorted(set(iris["species"])))
print("Counts:", iris.groupby("species").size().to_dict())
print("Matrix shape:", X.shape, "Numerical rank:", np.linalg.matrix_rank(X))
singular_values = np.linalg.svd(X, compute_uv=False)
plt.figure(figsize=(6, 4)); plt.plot(np.arange(1, 5), singular_values, marker="o")
plt.xlabel("Singular-value index"); plt.ylabel("Singular value")
plt.title("Iris measurement matrix — a linear-algebra diagnostic")
show_and_save(OUT / "iris_singular_values.png")
save_json(OUT / "results.json", {"real_rank": real_rank, "F2_rank": binary_rank,
          "iris_rank": int(np.linalg.matrix_rank(X)), "singular_values": singular_values})
