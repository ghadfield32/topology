1. Registration on known correspondences

Split points before fitting. We manufacture a similarity-transformed candidate from real reference-derived geometry, then fit only on anchor IDs. The withheld points check the algebra on unfit coordinates; they are not independent physical measurements.

Code · execution 3

from shape_lab.persistence import rips_filtration,persistent_homology,diagram,finite_bottleneck
cloud,rc=point_cloud(reference_disparity,calibration,stride=20)
chosen=rng.choice(len(cloud),size=120,replace=False);reference=cloud[chosen]
Rtrue=exp_so3([.12,-.2,.05]);scale_true=1.7;ttrue=np.array([.3,-.1,.8])
candidate=(reference-ttrue)@Rtrue/scale_true
anchor=np.arange(40);holdout=np.arange(40,120)
s,R,t=fit_similarity(candidate[anchor],reference[anchor])
registered=s*candidate@R.T+t
hold_rmse=float(np.sqrt(np.mean(np.sum((registered[holdout]-reference[holdout])**2,axis=1))))
assert hold_rmse<1e-12
print("Recovered similarity:",s,t,"held-out consistency RMSE (m):",hold_rmse)
perturbed=candidate.copy();perturbed[holdout]+=np.array([0,0,.03])
registered_bad=s*perturbed@R.T+t
bad_rmse=float(np.sqrt(np.mean(np.sum((registered_bad[holdout]-reference[holdout])**2,axis=1))))
assert bad_rmse>.01
print("Withheld-only defect detected, RMSE (m):",bad_rmse)
plt.figure(figsize=(6,5));plt.scatter(reference[:,0],reference[:,2],label="reference",s=12);plt.scatter(registered_bad[holdout,0],registered_bad[holdout,2],label="defective registered holdout",s=12)
plt.xlabel("x (m)");plt.ylabel("z (m)");plt.title("Registration cannot certify withheld geometry");plt.legend()
show_and_save(OUT / "registration_holdout.png")
