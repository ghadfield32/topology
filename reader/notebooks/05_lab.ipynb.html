from shape_lab.algebra import closure, euler_characteristic
examples = {"triangle_boundary": [(0,1), (0,2), (1,2)],
            "filled_triangle": [(0,1,2)],
            "tetrahedron_boundary": [(0,1,2),(0,1,3),(0,2,3),(1,2,3)],
            "solid_tetrahedron": [(0,1,2,3)]}
counts = {}
for name, facets in examples.items():
    complex_ = closure(facets)
    counts[name] = {"counts_by_dimension": [sum(len(s)==k+1 for s in complex_) for k in range(4)],
                    "euler": euler_characteristic(complex_)}
print(json.dumps(counts, indent=2))

Real pixels as closed squares

The triangulation preserves the selected union of squares. It is not a learned reconstruction of a hidden ideal digit.

Code · execution 4

from shape_lab.data import digit_example
from shape_lab.persistence import pixel_filtration
image, sample_id = digit_example(0)
f = pixel_filtration(1 - image.astype(float)/16)
threshold = .5
complex_ = [s for s, value in f if value <= threshold]
triangles = [s for s in complex_ if len(s)==3]
coords = np.array([(c, -r) for r in range(9) for c in range(9)])
plt.figure(figsize=(5, 5))
plt.triplot(coords[:, 0], coords[:, 1], np.array(triangles))
plt.axis("equal"); plt.title(f"Real training digit {sample_id}: selected closed-square triangulation")
plt.xlabel("Pixel-grid column"); plt.ylabel("Negative pixel-grid row")
show_and_save(OUT / "pixel_triangulation.png")
print("Threshold:", threshold, "Simplices:", len(complex_), "Euler:", euler_characteristic(complex_))
save_json(OUT / "results.json", {"exact_examples": counts, "real_sample_id": sample_id,
          "threshold": threshold, "simplex_count": len(complex_), "euler": euler_characteristic(complex_),
          "connectivity": "Closed squares; corner contact connects."})
