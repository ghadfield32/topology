Rips versus Čech on an exact equilateral triangle

This analytic example is not a general Čech implementation. The two parameters have different meanings.

Code · execution 3

from shape_lab.algebra import closure, betti_numbers
from shape_lab.persistence import rips_filtration, persistent_homology, diagram
triangle = np.array([[0.,0.], [1.,0.], [.5,np.sqrt(3)/2]])
# Use exact common event values for the analytic combinatorial Rips example.
# Floating geometric distances can differ by machine epsilon, so derive the shared value explicitly.
triangle_rips = [((i,), 0.) for i in range(3)] + [((0,1),1.),((0,2),1.),((1,2),1.),((0,1,2),1.)]
assert len(diagram(persistent_homology(triangle_rips), 1)) == 0
cech_rows = []
for radius in [.45, .51, .60]:
    facets = [(0,), (1,), (2,)]
    if 2*radius >= 1: facets += [(0,1),(0,2),(1,2)]
    if radius >= 1/np.sqrt(3): facets += [(0,1,2)]
    cech_rows.append({"ball_radius": radius, "betti": betti_numbers(closure(facets), max_dim=1)})
print("Čech analytic triangle:", cech_rows)
assert cech_rows[1]["betti"] == [1,1]
assert cech_rows[2]["betti"] == [1,0]

from shape_lab.data import load_iris_data
from shape_lab.display import plot_diagram
iris = load_iris_data(); selected = np.arange(0, 150, 10)
X = iris.iloc[selected, 1:5].to_numpy()
f = rips_filtration(X, max_homology=1)
print("Selected IDs:", selected, "Stored simplices:", len(f))
values = sorted(set(v for _,v in f))
thresholds = np.quantile(values, np.linspace(0,1,7))
rows=[]
for threshold in thresholds:
    b = betti_numbers([s for s,v in f if v<=threshold], max_dim=1)
    rows.append({"epsilon": float(threshold), "beta0": b[0], "beta1": b[1]})
print(pd.DataFrame(rows).to_string(index=False))
plt.figure(figsize=(6,4))
plt.plot([r["epsilon"] for r in rows], [r["beta0"] for r in rows], marker="o", label="beta0")
plt.plot([r["epsilon"] for r in rows], [r["beta1"] for r in rows], marker="s", label="beta1")
plt.xlabel("Rips maximum edge distance (cm)"); plt.ylabel("Betti number"); plt.legend()
plt.title("Real Iris teaching subset: scale changes the constructed complex")
show_and_save(OUT / "iris_filtration.png")
save_json(OUT / "results.json", {"cech_triangle": cech_rows, "iris_ids": selected,
          "rips_convention": "maximum edge length", "max_simplex_dimension": 2,
          "max_reported_homology": 1, "threshold_results": rows})
