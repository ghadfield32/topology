Real image, exact group action

Quarter-turns and reflections permute pixels without interpolation. We are not claiming the digit label is preserved.

Code · execution 3

from shape_lab.data import digit_example
image, sample_id = digit_example(2)
def R(a): return np.rot90(a)
def S(a): return np.fliplr(a)
assert np.array_equal(R(R(R(R(image)))), image)
assert np.array_equal(S(S(image)), image)
assert np.array_equal(S(R(S(image))), R(R(R(image))))
noncommuting = not np.array_equal(R(S(image)), S(R(image)))
print("Training sample ID:", sample_id, "Rotation/reflection differ by order:", noncommuting)
assert noncommuting
plt.figure(figsize=(4, 4)); plt.imshow(R(image)); plt.colorbar(label="Intensity")
plt.title(f"Quarter-turn of real training image {sample_id}; label not assumed invariant")
show_and_save(OUT / "rotated_digit.png")

Synthetic geometric control: an annulus deformation

The proof is in the lesson. This samples it and checks the radius bounds numerically.

Code · execution 4

angles = np.linspace(0, 2*np.pi, 40, endpoint=False)
radii = np.repeat([1., 1.5, 2.], len(angles))
angles = np.tile(angles, 3)
points = np.column_stack([radii*np.cos(angles), radii*np.sin(angles)])
def H(x, t):
    return ((1-t) + t/np.linalg.norm(x, axis=1))[:, None] * x
for t in np.linspace(0, 1, 11):
    norms = np.linalg.norm(H(points, t), axis=1)
    assert norms.min() >= 1-1e-12 and norms.max() <= 2+1e-12
assert np.allclose(np.linalg.norm(H(points, 1), axis=1), 1)
plt.figure(figsize=(5, 5))
plt.scatter(points[:, 0], points[:, 1], s=9, label="Original samples")
after = H(points, .5)
plt.scatter(after[:, 0], after[:, 1], s=9, label="Halfway through deformation")
plt.axis("equal"); plt.legend(); plt.title("Synthetic annulus: a sampled deformation retraction")
show_and_save(OUT / "annulus_deformation.png")
save_json(OUT / "results.json", {"real_sample_id": sample_id, "noncommuting_action": noncommuting,
          "annulus_sampled_checks": True, "proof_required_separately": True})
