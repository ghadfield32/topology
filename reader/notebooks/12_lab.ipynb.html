Code · execution 4

train_images,y_train,train_ids=load_digit_data("train")
val_images,y_val,val_ids=load_digit_data("validation")
test_images,y_test,test_ids=load_digit_data("test")
T_train,_,_=feature_data("train");T_val,_,_=feature_data("validation");T_test,_,_=feature_data("test")
y_dev=np.concatenate([y_train,y_val])
P_dev=np.concatenate([train_images,val_images]).reshape(len(y_dev),-1).astype(float)
P_test=test_images.reshape(len(test_images),-1).astype(float)
T_dev=np.vstack([T_train,T_val])
assert not (set(train_ids)|set(val_ids)) & set(test_ids)
representations={"pixels":(P_dev,P_test),"topology":(T_dev,T_test),
                 "combined":(np.column_stack([P_dev,T_dev]),np.column_stack([P_test,T_test]))}
results=[]; predictions=pd.DataFrame({"sample_id":test_ids,"true_label":y_test})
class_reports={}
for name,(X_dev,X_test) in representations.items():
    model=make_pipeline(StandardScaler(),LogisticRegression(C=selection["selected_C"][name],max_iter=3000))
    model.fit(X_dev,y_dev)
    prediction=model.predict(X_test)
    predictions[name]=prediction
    accuracy=float(accuracy_score(y_test,prediction))
    results.append({"representation":name,"test_accuracy":accuracy,
                    "correct":int(np.sum(prediction==y_test)),"test_count":len(y_test)})
    class_reports[name]=classification_report(y_test,prediction,output_dict=True,zero_division=0)
    matrix=confusion_matrix(y_test,prediction,labels=np.arange(10))
    plt.figure(figsize=(6,5));plt.imshow(matrix);plt.colorbar(label="Test images")
    plt.xticks(range(10));plt.yticks(range(10));plt.xlabel("Predicted digit");plt.ylabel("Recorded digit")
    plt.title(f"{name}: held-out image confusion matrix")
    show_and_save(OUT / f"confusion_{name}.png")
dummy=DummyClassifier(strategy="most_frequent").fit(P_dev,y_dev)
dummy_pred=dummy.predict(P_test)
results.append({"representation":"majority_baseline","test_accuracy":float(accuracy_score(y_test,dummy_pred)),
                "correct":int(np.sum(dummy_pred==y_test)),"test_count":len(y_test)})
print(pd.DataFrame(results).to_string(index=False))
predictions.to_csv(OUT / "test_predictions.csv",index=False)
save_json(OUT / "class_reports.json",class_reports)
save_json(OUT / "results.json",{"results":results,"selected_C":selection["selected_C"],
          "development_count":len(y_dev),"test_count":len(y_test),"fingerprint":fingerprint(),
          "claim":"Held-out image accuracy on the declared split; no unseen-writer or statistical-significance claim.",
          "holdout_status_after_run":"exposed"})
