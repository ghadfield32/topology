An independently coded H₀ merge calculation

This union-find procedure does not use persistence boundary reduction. It handles zero distances explicitly.

Code · execution 4

from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def merge_distances(points):
    distances = squareform(pdist(points))
    parent = list(range(len(points)))
    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i
    result=[]
    edges=sorted((distances[i,j],i,j) for i,j in combinations(range(len(points)),2))
    for distance,i,j in edges:
        a,b=find(i),find(j)
        if a != b:
            parent[b]=a
            result.append(float(distance))
    return result

from shape_lab.data import load_iris_data
iris=load_iris_data(); selected=np.arange(0,150,7)
X=iris.iloc[selected,1:5].to_numpy()
f=rips_filtration(X)
bars=persistent_homology(f)
ph_deaths=np.sort(diagram(bars,0,finite_only=True)[:,1])
merge_deaths=np.array([v for v in merge_distances(X) if v>0])
assert np.allclose(ph_deaths,merge_deaths)
checks=[]
for t in np.quantile(sorted(set(v for _,v in f)),[.15,.35,.6]):

    assert direct==via_bars
    checks.append({"threshold":float(t),"direct":direct,"barcode":via_bars})
