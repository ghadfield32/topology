rng=np.random.default_rng(29);middle=rng.permutation(np.arange(1,len(hahn)-1))
train=np.r_[0,len(hahn)-1,middle[:32]];validation=middle[32:43];test=middle[43:]
assert len(train)==34 and len(validation)==len(test)==11
T=hahn.temperature_K.to_numpy();y=hahn.expansion_response.to_numpy();mu=T[train].mean();scale=T[train].std();a=(T-mu)/scale
candidates=[];coefficients={}
for degree in [1,3,5]:
    X=np.polynomial.polynomial.polyvander(a,degree)
    theta=np.linalg.lstsq(X[train],y[train],rcond=None)[0];coefficients[degree]=theta
    candidates.append({'degree':degree,'validation_mae':float(np.mean(np.abs(X[validation]@theta-y[validation])))})
selected=min(candidates,key=lambda r:r['validation_mae'])['degree']
X=np.polynomial.polynomial.polyvander(a,selected);p=X[test]@coefficients[selected]
