cmake_minimum_required(VERSION 3.20)
project(shape_c LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(SHAPE_SANITIZE "Address and undefined-behavior sanitizers" OFF)
add_library(trajectory_reader_lib src/trajectory.c)
target_include_directories(trajectory_reader_lib PUBLIC include)
add_executable(trajectory_reader src/main.c)
target_link_libraries(trajectory_reader PRIVATE trajectory_reader_lib)
if(NOT MSVC)
 target_compile_options(trajectory_reader_lib PRIVATE -Wall -Wextra -Wpedantic)
 if(SHAPE_SANITIZE)
  foreach(t trajectory_reader_lib trajectory_reader)
   target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endforeach()
 endif()
endif()
include(CTest)
add_test(NAME rejects_missing_input COMMAND trajectory_reader)
set_tests_properties(rejects_missing_input PROPERTIES WILL_FAIL TRUE)

add_executable(memory_demo src/memory_demo.c)
add_test(NAME pointer_and_layout COMMAND memory_demo)
