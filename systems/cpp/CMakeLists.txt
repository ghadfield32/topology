cmake_minimum_required(VERSION 3.20)
project(shape_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SHAPE_SANITIZE "ASan+UBSan" OFF)
option(SHAPE_PYBIND "Build actual pybind11 extension" OFF)
option(SHAPE_OPENCV "Build C++ OpenCV experiment" OFF)
add_library(shape_core STATIC src/shape.cpp)
set_target_properties(shape_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(shape_core PUBLIC include)
add_executable(trajectory_cpp src/main.cpp)
target_link_libraries(trajectory_cpp PRIVATE shape_core)
add_executable(test_shape tests/test_core.cpp)
target_link_libraries(test_shape PRIVATE shape_core)
if(NOT MSVC)
 foreach(t shape_core trajectory_cpp test_shape)
  target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
  if(SHAPE_SANITIZE)
   target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endif()
 endforeach()
endif()
if(SHAPE_PYBIND)
 find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
 # Standard path: pip-installed pybind11 CMake config. Explicit headers path supports an inspected existing header distribution.
 if(SHAPE_PYBIND_HEADERS)
  Python_add_library(shape_cpp MODULE WITH_SOABI src/bindings.cpp)
  target_include_directories(shape_cpp PRIVATE "${SHAPE_PYBIND_HEADERS}")
 else()
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(shape_cpp src/bindings.cpp)
 endif()
 target_link_libraries(shape_cpp PRIVATE shape_core)
endif()
if(SHAPE_OPENCV)
 find_package(OpenCV REQUIRED COMPONENTS core calib3d)
 add_executable(opencv_probe src/opencv_probe.cpp)
 target_include_directories(opencv_probe PRIVATE ${OpenCV_INCLUDE_DIRS})
 target_link_libraries(opencv_probe PRIVATE ${OpenCV_LIBS})
endif()
include(CTest)
add_test(NAME math_contract COMMAND test_shape)
add_test(NAME rejects_missing_arguments COMMAND trajectory_cpp)
set_tests_properties(rejects_missing_arguments PROPERTIES WILL_FAIL TRUE)

add_executable(test_concepts tests/test_concepts.cpp)
add_test(NAME value_and_resource_concepts COMMAND test_concepts)
